### 2-2. 저장하지 않기 때문에 할 수 없는 것
<callout icon="⚠️" color="orange_bg">
	무저장 원칙의 대가를 문서에 남긴다. 아래는 구현 누락이 아니라 **설계상 불가능한 것**이며, 화면에서도 그렇게 안내한다. (FR-MO-008)
</callout>
- 공사 원문 텍스트 자체의 변경 전후 diff
- 판정 필드에 포함되지 않은 필드의 변경 감지
- 검수한 적 없는 기간의 이력 조회
- 과거 시점의 관광지 명칭 · 주소 · 이미지 복원
- 비표출로 전환된 콘텐츠의 과거 정보 표시

## 4. 테이블 상세 (DR-MD)
<callout icon="🧾" color="blue_bg">
	타입은 PostgreSQL 기준이다. 모든 테이블은 `id BIGSERIAL PRIMARY KEY`, `created_at timestamptz NOT NULL DEFAULT now()` 를 공통으로 갖는다(별도 표기 생략).
</callout>
```sql
-- 1. 계정
account(
  email            text NOT NULL UNIQUE,
  password_hash    text NOT NULL,
  is_demo          boolean NOT NULL DEFAULT false   -- 심사용 테스트 계정
)

-- 2. 상품
product(
  account_id       bigint NOT NULL REFERENCES account(id),
  name             text NOT NULL,
  ldong_regn_cd    text NOT NULL,        -- 시도. 길이 가정 금지 (세종 36110)
  ldong_signgu_cd  text,                 -- 시군구. 단층제 지역은 NULL
  start_date       date NOT NULL,
  nights           smallint NOT NULL DEFAULT 0,     -- 0 = 당일
  target_key       text,                 -- R10 프로파일 조회 키
  concept_key      text,
  head_count       int,
  transport        text NOT NULL,        -- CHARTER_BUS | CAR | PUBLIC_TRANSIT
  released_at      timestamptz           -- 출시 승인 시각. NULL = 미출시
)

-- 3. 일정 항목
itinerary_item(
  product_id       bigint NOT NULL REFERENCES product(id) ON DELETE CASCADE,
  day_no           smallint NOT NULL,
  seq              smallint NOT NULL,
  start_time       time NOT NULL,
  end_time         time,                 -- NULL = 기본 체류시간으로 보완
  end_time_source  text NOT NULL,        -- INPUT | DWELL_DEFAULT | DWELL_FALLBACK
  place_label      text NOT NULL,        -- 사용자가 입력한 장소명. 공사 원문 아님
  item_type        text NOT NULL,        -- SIGHT | MEAL | LODGING | REST | MOVE | FREE
  kto_content_id   text,                 -- 확정된 contentid. NULL = 미확정
  content_type_id  smallint,
  lcls_systm1      text, lcls_systm2 text, lcls_systm3 text,
  mapx             numeric(12,8), mapy numeric(12,8),
  match_status     text NOT NULL         -- CONFIRMED | EXCLUDED | PENDING
)

-- 4. 검수 작업 큐
audit_job(
  product_id       bigint NOT NULL REFERENCES product(id) ON DELETE CASCADE,
  status           text NOT NULL,        -- QUEUED | RUNNING | DONE | FAILED
  trigger_type     text NOT NULL,        -- INITIAL | MANUAL | PATCH | BATCH
  progress_done    int NOT NULL DEFAULT 0,
  progress_total   int NOT NULL DEFAULT 0,
  audit_run_id     bigint REFERENCES audit_run(id),   -- 마이그레이션 순서는 audit_run 이후
  error_code       text,
  finished_at      timestamptz
)

-- 5. 검수 실행 (불변 기록)
audit_run(
  product_id       bigint NOT NULL REFERENCES product(id) ON DELETE CASCADE,
  executed_at      timestamptz NOT NULL,
  ruleset_version  text NOT NULL,
  readiness_score  smallint,             -- NULL = 부분 검수
  is_partial       boolean NOT NULL DEFAULT false,
  target_count     smallint NOT NULL,    -- 대상 콘텐츠 수
  failed_count     smallint NOT NULL DEFAULT 0,
  blocker_cnt      smallint NOT NULL DEFAULT 0,
  error_cnt        smallint NOT NULL DEFAULT 0,
  warn_cnt         smallint NOT NULL DEFAULT 0,
  unverified_cnt   smallint NOT NULL DEFAULT 0,
  weight_snapshot  jsonb NOT NULL,       -- 산출 시점 가중치. 소급 변경 방지
  travel_seconds   int,                  -- 총 이동시간(초). NULL = 산출하지 않음 / 0 = 합이 0
  travel_meters    int                   -- 총 이동거리(m). 둘은 항상 같이 NULL 이거나 같이 값이다
)

-- 6. 발견된 문제
finding(
  audit_run_id     bigint NOT NULL REFERENCES audit_run(id) ON DELETE CASCADE,
  rule_code        text NOT NULL,        -- R01 .. R10
  rule_version     text NOT NULL,
  severity         text NOT NULL,        -- BLOCKER | ERROR | WARNING | UNVERIFIED
  reason_code      text NOT NULL,
  target_item_id   bigint REFERENCES itinerary_item(id) ON DELETE SET NULL,
  target_item_id2  bigint REFERENCES itinerary_item(id) ON DELETE SET NULL,  -- 구간 규칙(R03·R08)
  message          text NOT NULL,
  evidence         jsonb NOT NULL,       -- 판정 입력값(원문 아님). 규칙별 구조 자율
  requires_external boolean NOT NULL DEFAULT false,
  external_source  text,                 -- 지도 · 기상 제공자명
  patches          jsonb NOT NULL DEFAULT '[]',   -- 수정안 최대 3개
  dismissed_at     timestamptz,
  dismiss_reason   text,
  confirmed_at     timestamptz           -- 확인 필요 목록 체크. 점수 무영향
)

-- 7. 검수 지문
content_fingerprint(
  audit_run_id     bigint NOT NULL REFERENCES audit_run(id) ON DELETE CASCADE,
  kto_content_id   text NOT NULL,
  content_type_id  smallint NOT NULL,
  fetched_at       timestamptz NOT NULL,
  kto_modified_time text NOT NULL,       -- 원본 YYYYMMDDHHmmss 문자열
  show_flag        smallint NOT NULL,
  field_names      text[] NOT NULL,      -- 지문 생성에 쓴 필드명 목록
  field_hash       text NOT NULL,        -- sha256, 64자 hex
  normalized_json  jsonb,                -- 5장 스키마. 파싱 전면 실패 시 NULL
  parse_confidence text NOT NULL         -- CONFIRMED | ESTIMATED | UNPARSED
)

-- 8. 패치 이력
patch_application(
  product_id       bigint NOT NULL REFERENCES product(id) ON DELETE CASCADE,
  applied_at       timestamptz NOT NULL,
  applied_by       bigint NOT NULL REFERENCES account(id),
  selected_patches jsonb NOT NULL,
  before_snapshot  jsonb NOT NULL,       -- 일정 전체 스냅샷
  after_snapshot   jsonb NOT NULL,
  before_audit_run_id bigint REFERENCES audit_run(id),
  after_audit_run_id  bigint REFERENCES audit_run(id),
  reverted_at      timestamptz           -- 되돌리기 시각
)

-- 9. 알림
notification(
  product_id       bigint NOT NULL REFERENCES product(id) ON DELETE CASCADE,
  kind             text NOT NULL,        -- RISK | OPPORTUNITY
  match_condition  smallint NOT NULL,    -- 1..6 (F13 탐색 조건)
  kto_content_id   text,
  change_hash_from text, change_hash_to text,   -- 변경 근거 지문 (표시용)
  change_key       text NOT NULL,               -- 재노출 판정 키 (DB 명세서 v1.7)
  body             jsonb NOT NULL,
  dismissed_at     timestamptz,
  read_at          timestamptz
)

-- 10. 계정 설정
user_setting(
  account_id       bigint NOT NULL UNIQUE REFERENCES account(id),
  weights          jsonb NOT NULL,       -- {"BLOCKER":25,"ERROR":10,"WARNING":4,"UNVERIFIED":3}
  r07_span_hours   smallint NOT NULL DEFAULT 6,
  r07_meal_minutes smallint NOT NULL DEFAULT 60,
  r04_threshold    smallint NOT NULL DEFAULT 3,
  watch_keywords   text[] NOT NULL DEFAULT '{}'
)  -- 배치 시각 · 활성화 · 일일 호출 예산은 전역 system_setting으로 이동 (7-2절 · DR-CF-007)
```

## 5. 운영정보 정규화 스키마 (DR-NM)
<callout icon="🧬" color="blue_bg">
	**이 장이 이 문서의 본체다.** 자유 텍스트를 규칙엔진이 결정론적으로 평가할 수 있는 구조로 바꾸는 계약이며, 필드 정의 · 병합 규칙 · 평가 순서 · 신뢰도 산정 네 가지가 모두 정의되어야 R01이 재현 가능해진다.
</callout>

### 5-2. 전체 스키마
```json
{
  "schemaVersion": "1.0",
  "sourceFieldNames": ["restdateculture", "usetimeculture"],

  "alwaysOpen": false,
  "weeklyClosed": ["MON"],
  "nthWeekday": [{ "nth": 3, "day": "MON" }],
  "fixedClosed": ["01-01"],
  "holidayRule": ["LUNAR_NEW_YEAR", "CHUSEOK"],
  "conditionalRule": [
    { "kind": "HOLIDAY_NEXT_DAY", "appliesTo": ["MON"] }
  ],
  "partialClosed": [
    { "scope": "실내 전시실", "on": ["01-01", "LUNAR_NEW_YEAR", "CHUSEOK"] }
  ],

  "openHours": {
    "open": "09:00",
    "close": "18:00",
    "breaks": [],
    "admissionCutoff": "17:00"
  },
  "dayOfWeekHours": [
    { "days": ["MON"],
      "open": "07:40", "close": "16:00",
      "breaks": [], "admissionCutoff": "15:45" },
    { "days": ["WED", "THU", "FRI", "SAT", "SUN"],
      "open": "07:40", "close": "19:30",
      "breaks": [{ "from": "15:30", "to": "17:00" }], "admissionCutoff": "19:15" }
  ],
  "seasonalHours": [
    { "from": "03-01", "to": "10-31", "label": "하절기",
      "open": "09:00", "close": "18:00", "breaks": [], "admissionCutoff": null }
  ],

  "confidence": {
    "overall": "ESTIMATED",
    "byPath": {
      "weeklyClosed": "CONFIRMED",
      "conditionalRule": "ESTIMATED",
      "openHours": "CONFIRMED"
    }
  },
  "unparsed": [
    { "fragment": "점포별 상이함", "reason": "TARGET_VARIES", "affects": ["openHours"] }
  ]
}
```

### 5-4. 조각 병합 규칙
원문은 구분자로 분해된 뒤 조각별로 해석되고, 마지막에 하나의 객체로 병합된다. 병합에서 충돌이 생기는 경우가 실데이터의 기본형이므로 규칙을 명시한다.
<table fit-page-width="true" header-row="true">
<tr>
<td>ID</td>
<td>요구사항</td>
<td>등급</td>
</tr>
<tr>
<td>DR-NM-010</td>
<td>배열형 휴무 필드(`weeklyClosed` · `fixedClosed` · `holidayRule` · `nthWeekday`)는 조각별 결과를 **합집합**으로 병합해야 한다. 중복은 제거한다.</td>
<td>M</td>
</tr>
<tr>
<td>DR-NM-011</td>
<td>`alwaysOpen = true` 조각과 휴무 조각이 함께 나오면 **휴관 범위로 갈라 처리**해야 한다. ① 시설 전체 휴관이면 `alwaysOpen`을 false로 내리고 휴무 항목을 유지한다. ② 일부 시설만 휴관이면 `alwaysOpen`을 **true로 유지**하고 예외를 `partialClosed`에 담는다. 두 경우 모두 해당 경로 신뢰도를 `ESTIMATED`로 낮춰야 한다. (실측: `연중무휴(1월 1일/설날/추석 당일은 오죽헌만 개방, 실내 전시실 휴관)` → ②에 해당)</td>
<td>M</td>
</tr>
<tr>
<td>DR-NM-012</td>
<td>휴관 대상이 시설 전체가 아니라 일부로 명시되면 `partialClosed`에 담고 휴무 필드에는 넣지 않아야 한다. `partialClosed`만으로는 **R01 차단을 발생시키지 않으며** 주의로 판정한다.</td>
<td>M</td>
</tr>
<tr>
<td>DR-NM-013</td>
<td>같은 요일에 서로 다른 `open` · `close`를 주장하는 조각이 둘 이상이면, 해당 요일 항목을 만들지 않고 `unparsed`에 `reason=CONDITIONAL`로 기록해야 한다. 임의로 하나를 선택하지 않는다.</td>
<td>M</td>
</tr>
<tr>
<td>DR-NM-014</td>
<td>`unparsed.fragment`에는 **분해된 조각만** 담아야 하며 원문 전문을 복사하지 않아야 한다. 조각 길이가 200자를 넘으면 앞 200자로 절단한다. (DR-PR-003)<br/>**「원문 문장」의 범위 (구현 확정 2026.09.10)** — 금지 대상은 **문장·절**이다. 파서가 뽑아낸 **대상어**는 판독 산출물이므로 구조 필드에 담을 수 있다. `partialClosed.scope`(`실내 전시실`)가 그것이며 30자로 묶는다(`MAX_SCOPE_LENGTH`). 반대로 절을 통째로 담던 `conditionalRule.note`는 **없앴다** — 뜻은 `kind`로 만들고, 그것으로 안 되는 뉘앙스는 저장하지 않고 3단 병기가 실시간으로 보여준다 (이슈 #361)</td>
<td>M</td>
</tr>
<tr>
<td>DR-NM-015</td>
<td>시각 라벨 분기(FR-AU-014)로 얻은 `admissionCutoff`는 **같은 조각에서 나온 운영시간 항목에 귀속**시켜야 한다. 요일별 항목이 있으면 그 항목에, 없으면 `openHours`에 넣는다.</td>
<td>M</td>
</tr>
<tr>
<td>DR-NM-016</td>

### 5-5. 판정 평가 순서
<callout icon="⚙️" color="purple_bg">
	규칙엔진은 아래 순서를 **고정된 절차로** 수행한다. 순서가 바뀌면 같은 입력에 다른 결과가 나오므로, 이 절차 자체가 재현성의 정의다.
</callout>
```plain text
입력: 정규화 결과 N, 방문 날짜 D, 방문 시각 T

[1단계] 휴무 판정 — 먼저, 그리고 위에서부터. 하나라도 걸리면 즉시 CLOSED
  1-1  N.fixedClosed 에 D의 MM-DD 가 있는가              → CLOSED (확정)
  1-2  N.holidayRule 의 명절이 D에 해당하는가            → CLOSED (확정)
  1-3  N.nthWeekday 중 D가 매월 N번째 해당 요일인가       → CLOSED (확정)
  1-4  N.weeklyClosed 에 D의 요일이 있는가               → CLOSED (확정)
  1-5  N.conditionalRule 이 D에 적용될 여지가 있는가      → CLOSED_UNCERTAIN (추정)
  1-6  위 전부 해당 없음 그리고 N.alwaysOpen = true       → OPEN
  1-7  위 전부 해당 없음 그리고 alwaysOpen = false 이며
       휴무 필드가 모두 비어 있음                       → UNKNOWN

[2단계] 운영시간 선택 — 구체성이 높은 것이 이긴다
  2-1  N.dayOfWeekHours 에서 D의 요일이 속한 항목        → 채택
  2-2  없으면 N.seasonalHours 에서 D가 속한 기간 항목     → 채택
  2-3  없으면 N.openHours                               → 채택
  2-4  아무것도 없으면                                   → UNKNOWN
  ※ 2-1과 2-2가 동시에 매칭되고 open·close가 상충하면
     2-1을 채택하되 해당 경로 신뢰도를 ESTIMATED 로 낮춘다

[3단계] 시각 판정 — 채택된 항목 H에 대해
  3-1  T < H.open 또는 T >= H.close                     → OUT_OF_HOURS
  3-2  T 가 H.breaks 의 어느 구간에 포함                  → IN_BREAK
  3-3  H.admissionCutoff 가 있고 T > cutoff              → AFTER_CUTOFF
  3-4  그 외                                            → OPEN

[4단계] 등급 매핑
  CLOSED            → 차단        (사유코드 REST_DAY_CONFLICT)
  OUT_OF_HOURS      → 차단        (사유코드 OPEN_HOUR_CONFLICT)
  AFTER_CUTOFF      → 오류        (사유코드 ADMISSION_CUTOFF)
  IN_BREAK          → 주의        (사유코드 IN_BREAK_TIME)
  CLOSED_UNCERTAIN  → 주의 + 확인 필요 목록 동시 등록 (사유코드 REST_DAY_UNCERTAIN)
  UNKNOWN           → 확인 불가   (R05로 이관)
  OPEN              → finding 없음

[5단계] 신뢰도 게이트 — 4단계 결과에 덮어쓰기
  판정에 사용한 필드 경로의 신뢰도가 ESTIMATED 이면
  차단 등급을 주의로 강등하고 확인 필요 목록에 동시 등록한다  (FR-AU-008)
  UNPARSED 이면 확인 불가로 강등한다
```

### 5-7. 실측 원문 변환 예시
세 건 모두 2026.08.12 – 13 실호출에서 그대로 가져온 원문이다. 스키마가 실데이터를 표현할 수 있는지 검증하는 기준 사례이며, 회귀 테스트의 골든 케이스가 된다.
**사례 1 — 강릉 오죽헌·시립박물관** · contentTypeId 14 · contentid 129784
```plain text
restdateculture = 연중무휴(1월 1일/설날/추석 당일은 오죽헌만 개방, 실내 전시실 휴관)
usetimeculture  = 매표시간 09:00~17:00 관람시간 09:00~18:00
```
```json
{
  "schemaVersion": "1.0",
  "sourceFieldNames": ["restdateculture", "usetimeculture"],
  "alwaysOpen": true,
  "weeklyClosed": [], "nthWeekday": [], "fixedClosed": [], "holidayRule": [],
  "conditionalRule": [],
  "partialClosed": [
    { "scope": "실내 전시실", "on": ["01-01", "LUNAR_NEW_YEAR", "CHUSEOK"] }
  ],
  "openHours": {
    "open": "09:00", "close": "18:00",
    "breaks": [], "admissionCutoff": "17:00"
  },
  "dayOfWeekHours": [], "seasonalHours": [],
  "confidence": {
    "overall": "ESTIMATED",
    "byPath": { "alwaysOpen": "ESTIMATED", "partialClosed": "ESTIMATED", "openHours": "CONFIRMED" }
  },
  "unparsed": []
}
```
> **판정 결과** — 10/13(월) 10:00 방문 시 1단계 1-1 – 1-5 전부 미해당, 1-6에서 `alwaysOpen = true` → `OPEN`. 2단계 `openHours` 채택, 3단계 `09:00 ≤ 10:00 < 18:00` 이고 cutoff 17:00 이전 → **finding 없음.**
> 괄호 예외가 시설 **일부**에만 걸리므로 `alwaysOpen`은 `true`로 유지하고 예외는 `partialClosed`가 담는다(DR-NM-011). 1/1 · 설날 · 추석 방문이면 주의 finding이 별도 생성된다(DR-NM-022).
> **이 사례가 잡아낸 것** — 라벨 분기가 없었다면 폐장을 17:00으로 오판해 **없는 차단**을 만들었을 것이다.
**사례 2 — 소나무집초당순두부** · contentTypeId 39 · contentid 134338

```plain text
opentimefood = [월요일]- 07:40~16:00- 마지막 주문 15:45[수요일~일요일]- 07:40~19:30- 준비시간 15:30~17:00- 마지막 주문 19:15
```

```plain text

  P.field_hash = C.field_hash          → 변경 없음. 직전 판정 재사용 가능
```
